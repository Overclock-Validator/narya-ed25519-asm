# Build baseline and candidate in separate clean worktrees.
make clean
make -j2 CFLAGS="-O3 -g -std=c11 -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror" test-native
cc -O3 -std=c11 -Wall -Wextra -Wpedantic -Werror -Iinclude bench_verify.c build/libnarya_ed25519_asm.a -o bench-verify

# Alternate order for ten samples; each process performs 256 warm-up calls.
taskset -c 2 ./bench-verify tests/vectors/narya_strict_verify_v1.txt 20000 1

# Counter run, repeated five times for each exact binary.
perf stat -x, -e cycles,instructions,task-clock -r 5 \
  taskset -c 2 ./bench-verify tests/vectors/narya_strict_verify_v1.txt 10000 1

# Candidate release gates.
make clean
make -j2 CFLAGS="-O3 -g -std=c11 -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror" test-native
make check-r51-object-bytes
